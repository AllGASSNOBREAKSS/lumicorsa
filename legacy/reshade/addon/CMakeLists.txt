cmake_minimum_required(VERSION 3.20)
project(photolap_addon CXX)
set(CMAKE_CXX_STANDARD 20)
add_library(photolap SHARED src/photolap_addon.cpp)
target_include_directories(photolap PRIVATE ../../../third_party/reshade/include ../../../third_party/imgui)
set_target_properties(photolap PROPERTIES PREFIX "" SUFFIX ".addon64" OUTPUT_NAME "photolap")
if(MSVC)
  target_compile_options(photolap PRIVATE /W3 /O2 /EHsc)
  set_property(TARGET photolap PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
else()
  target_compile_options(photolap PRIVATE -fpermissive -w)
  target_link_options(photolap PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_libraries(photolap PRIVATE psapi)
endif()
